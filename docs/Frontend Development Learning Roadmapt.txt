```jsx
import mapboxgl from 'mapbox-gl';
import 'mapbox-gl/dist/mapbox-gl.css';

function RailwayMap() {
  const mapContainer = useRef(null);
  const map = useRef(null);
  
  useEffect(() => {
    // 1. Initialize Map with Dark Theme
    mapboxgl.accessToken = 'YOUR_MAPBOX_TOKEN';
    map.current = new mapboxgl.Map({
      container: mapContainer.current,
      style: 'mapbox://styles/mapbox/dark-v11', // Dark theme
      center: [88.35, 22.58], // Howrah coordinates
      zoom: 11,
      pitch: 45 // 3D tilt
    });
    
    // 2. Add Railway Track Line
    map.current.on('load', () => {
      map.current.addSource('railway-track', {
        type: 'geojson',
        data: {
          type: 'Feature',
          geometry: {
            type: 'LineString',
            coordinates: [
              [88.35, 22.58], // Howrah
              [88.30, 22.55], // Santragachi
              [88.15, 22.40]  // Kharagpur
            ]
          }
        }
      });
      
      map.current.addLayer({
        id: 'track-line',
        type: 'line',
        source: 'railway-track',
        paint: {
          'line-color': '#00ff88', // Green for free section
          'line-width': 5,
          'line-blur': 2
        }
      });
      
      // 3. Add Station Markers
      const stations = [
        { name: 'Howrah Jn', coords: [88.35, 22.58], type: 'major' },
        { name: 'Santragachi', coords: [88.30, 22.55], type: 'minor' },
        { name: 'Kharagpur', coords: [88.15, 22.40], type: 'major' }
      ];
      
      stations.forEach(station => {
        new mapboxgl.Marker({
          color: station.type === 'major' ? '#ff4444' : '#4488ff'
        })
          .setLngLat(station.coords)
          .setPopup(new mapboxgl.Popup().setHTML(`
            <strong>${station.name}</strong><br>
            Status: Operational
          `))
          .addTo(map.current);
      });
      
      // 4. Animate Train Movement
      const train = new mapboxgl.Marker({ color: '#ffd700' })
        .setLngLat([88.32, 22.56])
        .addTo(map.current);
      
      let progress = 0;
      function animateTrain() {
        progress += 0.003;
        if (progress > 1) progress = 0;
        
        // Interpolate position along track
        const lng = 88.35 + (88.15 - 88.35) * progress;
        const lat = 22.58 + (22.40 - 22.58) * progress;
        
        train.setLngLat([lng, lat]);
        requestAnimationFrame(animateTrain);
      }
      animateTrain();
    });
    
  }, []);
  
  return <div ref={mapContainer} style={{ width: '100%', height: '600px' }} />;
}
```

```javascript
// hooks/useCorridorSocket.js
import { useEffect, useRef } from 'react';
import { useQueryClient } from '@tanstack/react-query';

export function useCorridorSocket(corridorCode) {
  const ws = useRef(null);
  const queryClient = useQueryClient();
  
  useEffect(() => {
    const token = localStorage.getItem('access_token');
    ws.current = new WebSocket(
      `ws://localhost:8001/ws/v1/corridor/${corridorCode}/?token=${token}`
    );
    
    ws.current.onopen = () => {
      console.log('WebSocket connected');
    };
    
    ws.current.onmessage = (event) => {
      const data = JSON.parse(event.data);
      
      if (data.type === 'INVALIDATE_CACHE') {
        // Invalidate TanStack Query cache
        queryClient.invalidateQueries({ 
          queryKey: ['blocks', data.entity_id] 
        });
      }
      
      if (data.type === 'EMERGENCY_ALERT') {
        // Show emergency modal
        useUIStore.getState().setEmergencyAlert(data.payload);
      }
    };
    
    ws.current.onclose = () => {
      console.log('WebSocket disconnected');
      // Reconnection logic
      setTimeout(() => {
        // Reconnect after 3 seconds
      }, 3000);
    };
    
    return () => {
      ws.current?.close();
    };
  }, [corridorCode]);
}
```

### How It Will Be Used in Your Project:

```javascript
// hooks/useBlocks.js
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import api from '../services/api';

export function useBlocks(corridorId) {
  return useQuery({
    queryKey: ['blocks', corridorId],
    queryFn: async () => {
      const response = await api.get(`/blocks/?corridor=${corridorId}`);
      return response.data.data;
    },
    staleTime: 30000, // 30 seconds
    cacheTime: 5 * 60 * 1000, // 5 minutes
  });
}

export function useSanctionBlock() {
  const queryClient = useQueryClient();
  
  return useMutation({
    mutationFn: (blockId) => api.post(`/blocks/${blockId}/sanction/`),
    onSuccess: () => {
      // Invalidate blocks cache
      queryClient.invalidateQueries({ queryKey: ['blocks'] });
    },
  });
}

// Component usage
function BlockList({ corridorId }) {
  const { data: blocks, isLoading } = useBlocks(corridorId);
  const sanctionMutation = useSanctionBlock();
  
  if (isLoading) return <LoadingSpinner />;
  
  return (
    <div>
      {blocks.map(block => (
        <BlockCard 
          key={block.id} 
          block={block}
          onSanction={() => sanctionMutation.mutate(block.id)}
        />
      ))}
    </div>
  );
}
```
